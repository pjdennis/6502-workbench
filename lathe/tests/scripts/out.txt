; Scripts for lathe/src/outtest.lth and its twin in lathe/tests/test_src_out.py, which write
; an object file (its name is the program's second argument) and compare it byte for byte.
; open; module CPU NAME; import NAME; sym KIND MODULE NAME; sym3 KIND MODULE INNER NAME;
; var SYMBOL SIZE CLASS WEIGHT; data SYMBOL LENGTH; bytes B...; table LENGTH;
; tset OFFSET BYTE; tfix OFFSET KIND SYMBOL SLOT ADDEND; tend SYMBOL LENGTH;
; call CALLEE FLAGS OUTER; class SYMBOL FLAGS COUNT; param SIZE; member CLASS FUNCTION;
; func SYMBOL FLAGS; slot SIZE CLASS WEIGHT KIND NAME; fend; end;
; ksym SCOPE KIND NAME OBJECT WEIGHT (a kernel symbol); fbegin; op OPCODE;
; fix OPCODE KIND SYMBOL SLOT ADDEND; finish OWN TEMP_BASE (FBUF).
; Symbol kinds: 1 function, 2 variable, 3 data, 4 class. Kernel kinds: 1 module, 2 function,
; 4 variable. Scopes: 0 this module, 1 and up imported, $FE predeclared, $FF local.
== a whole module: every record the writer has
open

import util
import a_module_with_a_long_name_xyz

sym 2 demo count
sym 3 demo $1
sym3 1 demo main inner_thing
sym 4 demo handler
sym 1 util put
sym 2 util a_module_with_a_long_name_xyz
sym3 2 aaaaaaaaaaaaaaaaaaaaaaaaaaaaaaa bbbbbbbbbbbbbbbbbbbbbbbbbbbbbbb ccccccccccccccccccccccccccccccc
var 1 2 1 5
var 7 512 0 255
var $FFFF $FFFF 2 0
data 2 5
bytes 1 2 3 4 255
data 6 0
table 6                     ; a table with fix-ups: DFIX first, in order of offset
tset 0 $11
tset 5 $55
tfix 3 2 0 $FF 4
tfix 1 3 0 $FF 0
tfix 4 3 0 $FF 4
tfix 2 2 0 $FF 0
tend 2 6
table 3                     ; and one without
tset 2 7
tend 3 3
class 4 1 3
param 1
param 2
param 512
class 4 0 0
member 4 0
member 4 0                  ; written once
member 4 1
member 5 0
call 5 0 $FFFF              ; calls are held until the function ends, each once
call 5 0 $FFFF
call 5 1 $FFFF
call 5 1 3
call 0 2 $FFFF
fbegin
fix $AD 1 0 0 0
fix $8D 1 0 $80 0
fix $AD 1 5 2 0             ; another function's slot: a weight
fix $AD 1 5 2 0
fix $20 1 5 $FF 0
fix $A9 2 1 $FF 1
fix $A5 4 0 1 0
op $60
finish 0 3
func 0 5
slot 1 1 0 0 a
slot 2 0 17 1 bb
slot 300 2 255 2 $t1
fend
fbegin
op $60
finish 3 0
func 3 0
fend                        ; no calls, fix-ups or weights
ksym 1 4 seen 6 9           ; a variable of another module that was used: a weight
ksym 1 4 unseen 7 0         ; no weight: not written
ksym 1 4 noobj $FFFF 3      ; no object symbol: not written
ksym 0 4 own 8 4            ; this module's: not written
ksym 1 2 func 9 4           ; not a variable
ksym 2 4 other 10 200
end
