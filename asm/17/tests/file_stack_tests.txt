# File Stack Component Tests
#
# Format:
#   NAME: test identifier
#   MODE: echo|lines|nested|info
#   FILE <filename>: (multi-line content, "N: " prefix stripped)
#   EXPECT_STDOUT: (multi-line expected output)
#   EXPECT_STDERR: (multi-line expected stderr, for error tests)
#
# The first FILE defined is the main input file.
# For nested mode, use @include <filename> to include other files.

# ============================================================================
# ECHO MODE - Basic character reading
# ============================================================================

---
NAME: echo_simple
MODE: echo
FILE input.txt:
 1: Hello
EXPECT_STDOUT:
Hello

---
NAME: echo_multiline
MODE: echo
FILE input.txt:
 1: Line1
 2: Line2
 3: Line3
EXPECT_STDOUT:
Line1
Line2
Line3

---
NAME: echo_empty
MODE: echo
FILE input.txt:
EXPECT_STDOUT:

---
NAME: echo_single_char
MODE: echo
FILE input.txt:
 1: X
EXPECT_STDOUT:
X

---
NAME: echo_no_trailing_newline
MODE: echo
FILE input.txt:
 1: NoNewline
EXPECT_STDOUT:
NoNewline

# ============================================================================
# NESTED MODE - Include handling
# ============================================================================

---
NAME: nested_no_includes
MODE: nested

Before

After

---

 3: B

 5: C

FILE two.txt:
 1: TWO

ONE
B
TWO
C

---
NAME: nested_deep

 1: Outer1
 2: @include middle.txt
 3: Outer2

 1: Middle1

 3: Middle2

 1: Inner

Middle1

Middle2
Outer2

NAME: nested_at_not_include

# ============================================================================
# MEMORY MODE - Memory source testing
# ============================================================================
# Tests @memory <content> which pushes content as a memory source
# When memory source is exhausted, reading continues from the file

---
NAME: memory_simple
MODE: memory
FILE input.txt:
 1: Before
 2: @memory INLINE
 3: After
EXPECT_STDOUT:
Before
INLINE
After

---
NAME: memory_multichar
MODE: memory
FILE input.txt:
 1: A
 2: @memory XYZ123
 3: B
EXPECT_STDOUT:
A
XYZ123
B

---
NAME: memory_empty
MODE: memory
FILE input.txt:
 1: Start
 2: @memory
 3: End
EXPECT_STDOUT:
Start

End

---
NAME: memory_multiple
MODE: memory
FILE input.txt:
 1: A
 2: @memory ONE
 3: B
 4: @memory TWO
 5: C
EXPECT_STDOUT:
A
ONE
B
TWO
C

---
NAME: memory_with_file_include
MODE: memory
FILE main.txt:
 1: File
 2: @memory MEM
 3: @include other.txt
 4: Done
FILE other.txt:
 1: Included
EXPECT_STDOUT:
File
MEM
Included
Done

# ============================================================================
# TRACEBACK MODE - Stack traceback testing
# ============================================================================
# Tests @traceback which prints the file stack and pops all entries

---
NAME: trace_simple
MODE: memory
FILE main.txt:
 1: Before
 2: @traceback
 3: After
EXPECT_STDOUT:
Before
file:main.txt:2

# Note: After @traceback pops the stack, reading continues but finds EOF
# so "After" is never printed

---
NAME: trace_nested
MODE: memory
FILE main.txt:
 1: Main start
 2: @include inner.txt
 3: Main end
FILE inner.txt:
 1: Inner start
 2: @traceback
 3: Inner end
EXPECT_STDOUT:
Main start
Inner start
file:inner.txt:2
file:main.txt:2

---
NAME: trace_deep
MODE: memory
FILE main.txt:
 1: Outer1
 2: @include middle.txt
 3: Outer2
FILE middle.txt:
 1: Middle1
 2: @include inner.txt
 3: Middle2
FILE inner.txt:
 1: Inner
 2: @traceback
EXPECT_STDOUT:
Outer1
Middle1
Inner
file:inner.txt:2
file:middle.txt:2
file:main.txt:2

---
NAME: trace_with_memory
MODE: memory
FILE main.txt:
 1: File start
 2: @memory MEM_CONTENT
 3: @traceback
 4: After trace
EXPECT_STDOUT:
File start
MEM_CONTENT
file:main.txt:2

# Note: Line shows 2 (where @memory was) because after memory exhausts,
# we restore the line from when memory was pushed

---
NAME: trace_alternating
MODE: memory
FILE main.txt:
 1: Main line 1
 2: @include inner.txt
 3: Main line 3
FILE inner.txt:
 1: Inner line 1
 2: @memory @traceback
 3: Inner line 3
EXPECT_STDOUT:
Main line 1
Inner line 1
memory:MEMORY:1
file:inner.txt:2
file:main.txt:2

# This tests file -> file -> memory stack with traceback showing all types
# The @traceback is inside the memory content, so it triggers while memory source is active
