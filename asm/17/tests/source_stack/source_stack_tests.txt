# ============================================================================
# TOP FRAME SIZE - Byte-level frame layout visibility
# ============================================================================
# @top_frame_size prints the size in bytes of the topmost source-stack frame.
# Post-reorg frame size is parent-type-independent:
#   1(size) + 1(curr) + 1(prev) + 2(line) + 2(prev_data, fixed)
#   + name_len + 1(null) + payload
# = name_len + 8 + payload regardless of parent. File parents waste 1
# byte (offset 6 unused) in exchange for fixed name offset and a
# constant frame-size formula.
# These tests pin the layout of file/memory frames so Phase 3's payload
# extension has an explicit byte budget to work against.

---
NAME: top_frame_size_memory_above_file
MODE: frames
FILE main.txt:
 1: @include inner.txt
FILE inner.txt:
 1: @memory @top_frame_size
EXPECT_STDOUT:
14

---
NAME: top_frame_size_include_above_file
MODE: frames
FILE main.txt:
 1: @include inner.txt
FILE inner.txt:
 1: @top_frame_size
EXPECT_STDOUT:
17

---
NAME: top_frame_size_memory_above_memory
MODE: frames
FILE main.txt:
 1: @include inner.txt
FILE inner.txt:
 1: @memory @memory @top_frame_size
EXPECT_STDOUT:
14

# Outer @memory pushes a frame whose content is the rest of line 1, which
# is "@memory @top_frame_size\n". When the outer memory source is read, an
# inner @memory directive fires and pushes a second memory frame containing
# "@top_frame_size\n". When @top_frame_size runs the inner memory frame
# is on top, name "MEMORY" (6) so size = 6+1+1+1+2+2+1 = 14. Memory and
# file parents now produce the same size (both 14 here).

# ============================================================================
# Phase 3.1 / 3.2: payload-bearing memory frames
# ============================================================================
# @payload_memory <N> pushes a memory source named PAYLOAD with N bytes of
# trailing payload, prints "ps:<frame_size>", then pops. Frame layout:
#   1 (size byte) + 1 (curr_type) + 1 (prev_type) + 2 (line)
#   + 2 (prev_data, fixed) + 7 (name "PAYLOAD") + 1 (null) + N (payload)
# = 15 + N regardless of parent type.
# Verifies push_memory_source_reserve_payload sizes the frame correctly.

---
NAME: payload_memory_zero
MODE: frames
FILE input.txt:
 1: @payload_memory 0
EXPECT_STDOUT:
ps:15

---
NAME: payload_memory_one
MODE: frames
FILE input.txt:
 1: @payload_memory 1
EXPECT_STDOUT:
ps:16

---
NAME: payload_memory_four
MODE: frames
FILE input.txt:
 1: @payload_memory 4
EXPECT_STDOUT:
ps:19

---
NAME: payload_memory_nine
MODE: frames
FILE input.txt:
 1: @payload_memory 9
EXPECT_STDOUT:
ps:24

---
NAME: payload_memory_round_trip
MODE: frames
FILE input.txt:
 1: @frames
 2: @payload_memory 5
 3: @frames
EXPECT_STDOUT:
0:file:input.txt
ps:20
0:file:input.txt
# Push then pop leaves the chain back at just the parent file frame --
# proves the round-trip works for payload-bearing frames.

---
NAME: payload_memory_above_memory
MODE: frames
FILE main.txt:
 1: @memory @payload_memory 3
EXPECT_STDOUT:
ps:18
# Memory parent: 15 + 3 = 18 (post-reorg; parent type no longer matters).

# ============================================================================
# Phase 5: ss_reserve_frame / ss_commit_pending_frame primitives
# ============================================================================
# @reserve_only reserves a memory frame named "PENDING" (no payload),
# verifies SS_P16 still reflects the parent (i.e. the reservation is
# invisible to consumers reading SS_P16), then discards the
# reservation. Prints "ps:<committed_top_frame_size>".
#
# @reserve_commit_pop reserves+commits a memory frame named
# "RESERVED" (no payload), prints "ps:<frame_size>", then pops. The
# round-trip should produce the same observable behaviour as an
# atomic push+pop pair.

---
NAME: reserve_only_invisible_to_top_frame_size
MODE: frames
FILE main.txt:
 1: @include inner.txt
FILE inner.txt:
 1: @reserve_only
EXPECT_STDOUT:
ps:17
# inner.txt is loaded via @include so its frame name is the literal
# "inner.txt" (9 chars), giving frame_size = 9 + 8 = 17. (The
# top-level main.txt path is the tmpdir-mangled full path and would
# break a same-frame size assertion.)
# After ss_reserve_frame, SS_P16 still points at inner.txt's frame.
# The reserved pending frame ("PENDING", 7 chars, size 15) lives
# below SS_PEND_P16 but is invisible -- ps:17 not ps:15 confirms
# invisibility.

---
NAME: reserve_only_invisible_to_frames
MODE: frames
FILE main.txt:
 1: @include inner.txt
FILE inner.txt:
 1: @reserve_only
 2: @frames
EXPECT_STDOUT:
ps:17
0:file:inner.txt
1:file:main.txt
# After @reserve_only discards its reservation, @frames walks the
# committed chain (inner.txt then main.txt). The reserved frame
# never appears in the @frames output even momentarily because
# @frames walks SS_P16, not SS_PEND_P16.

---
NAME: reserve_commit_pop_round_trip
MODE: frames
FILE input.txt:
 1: @reserve_commit_pop
 2: @frames
EXPECT_STDOUT:
ps:16
0:file:input.txt
# "RESERVED" is 8 chars; new memory frame is 8 + 8 + 0 = 16 bytes.
# After commit the new top is the RESERVED memory frame; after pop
# we're back to the parent (just input.txt in the chain).

---
NAME: reserve_commit_pop_matches_atomic_push_size
MODE: frames
FILE input.txt:
 1: @reserve_commit_pop
 2: @payload_memory 1
EXPECT_STDOUT:
ps:16
ps:16
# RESERVED is 8 chars + 0 payload = 16; PAYLOAD is 7 chars + 1 payload
# = 16. Pins down "frame size formula is parent-type-independent and
# the reserve/commit path produces frames structurally identical to
# the atomic push path."
