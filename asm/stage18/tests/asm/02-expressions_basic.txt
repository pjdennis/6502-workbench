# --- Expressions (expressions.asm) ---
# Expression evaluation, shifts, byte selectors, spaces in expressions, character literals, decimal constants

---
NAME: hex_single_char
INPUT:
 1: * = $0200
 2: foo = $7
 3:   LDA #foo
EXPECT_HEX: a9 07

---
NAME: hex_three_chars
INPUT:
 1: * = $0200
 2: foo = $712
 3:   LDA #<foo
 4:   LDA #>foo
EXPECT_HEX: a9 12 a9 07

---
NAME: expression_assignment
INPUT:
 1: * = $0200
 2: addr:  = $1234
 3: add:   = $01
 4: addr2: = addr+add
 3:   LDA addr
 4:   STA addr2
EXPECT_HEX: ad 34 12 8d 35 12

---
NAME: assignment_byte_select_multiple
INPUT:
 1: * = $0200
 2: val1: = $1234
 3: val2: = $5678
 4: low1: = <val1
 5: high1: = >val1
 6: low2: = <val2
 7: high2: = >val2
 8:   LDA #low1
 9:   LDX #high1
10:   LDY #low2
11:   .word high2
EXPECT_HEX: a9 34 a2 12 a0 78 56 00
EXPECT_FWDREF: 0
