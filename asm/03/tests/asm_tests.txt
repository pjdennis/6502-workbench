# Test suite for asm03 assembler
# Non-standard syntax: LDA#, STAZ, etc. PC starts at $2000.
# Features: labels, label assignment (=), DATA, < > byte selectors, strings, origin (* = $XXXX)
# Note: Single pass - no forward references for labels used as operands
# Note: Labels emit 2 bytes - only use with 2-byte operand instructions (STA, JMP, JSR)
# Note: Branches use raw hex offsets, not labels
# Note: Label assignment requires 4-char hex: label = $XXXX

---
NAME: implied_rts
INPUT:
1:   RTS
EXPECT_HEX: 60

---
NAME: immediate
INPUT:
1:   LDA# $42
2:   LDX# $00
3:   LDY# $FF
EXPECT_HEX: a9 42 a2 00 a0 ff

---
NAME: zero_page_hex
INPUT:
1:   LDAZ $42
2:   STAZ $10
EXPECT_HEX: a5 42 85 10

---
NAME: absolute
INPUT:
1:   STA $1234
EXPECT_HEX: 8d 34 12

---
NAME: zero_page_x_hex
INPUT:
1:   STAZ,X $20
EXPECT_HEX: 95 20

---
NAME: indirect_y_hex
INPUT:
1:   LDA(),Y $10
2:   STA(),Y $20
EXPECT_HEX: b1 10 91 20

---
NAME: data_hex
INPUT:
1:   DATA $01 $02 $03
EXPECT_HEX: 01 02 03

---
NAME: data_string
INPUT:
1:   DATA "Hi" $00
EXPECT_HEX: 48 69 00

---
NAME: data_mixed
INPUT:
1:   DATA $01 "AB" $02
EXPECT_HEX: 01 41 42 02

---
NAME: branch_with_hex_offset
INPUT:
1:   BEQ $01
2:   INX
3:   RTS
EXPECT_HEX: f0 01 e8 60

---
NAME: label_assignment_absolute
DESCRIPTION: Label assignment with 4-char hex, used with absolute instruction
INPUT:
1: addr = $1234
2:   STA addr
EXPECT_HEX: 8d 34 12

---
NAME: byte_selector_low
INPUT:
1: addr = $1234
2:   LDA# <addr
EXPECT_HEX: a9 34

---
NAME: byte_selector_high
INPUT:
1: addr = $1234
2:   LDA# >addr
EXPECT_HEX: a9 12

---
NAME: jmp_hex

---
NAME: jsr_label

2:   JSR sub

---
NAME: comment

---
NAME: data_label_bytes
