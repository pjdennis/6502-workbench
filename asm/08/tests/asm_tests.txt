# Test suite for asm08 assembler
# Non-standard syntax with hash table instruction lookup, two-pass assembly
# Features: labels with forward refs, branches with labels, DATA, < > selectors
# Has: origin (*=), label assignments, character literals ('A')

---
NAME: implied_rts
INPUT:
1: * = $0200
2:   RTS
EXPECT_HEX: 60

---
NAME: immediate
INPUT:
1: * = $0200
2:   LDA# $42
3:   LDX# $00
4:   LDY# $FF
EXPECT_HEX: a9 42 a2 00 a0 ff

---
NAME: zero_page
INPUT:
1: * = $0200
2:   LDAZ $42
3:   STAZ $10
EXPECT_HEX: a5 42 85 10

---
NAME: absolute
INPUT:
1: * = $0200
2:   STA $1234
EXPECT_HEX: 8d 34 12

---
NAME: zero_page_x
INPUT:
1: * = $0200
2:   LDAZ,X $20
3:   STAZ,X $30
EXPECT_HEX: b5 20 95 30

---
NAME: absolute_indexed
INPUT:
1: * = $0200
2:   LDA,X $1234
3:   LDA,Y $5678
EXPECT_HEX: bd 34 12 b9 78 56

---
NAME: indirect_y
INPUT:
1: * = $0200
2:   LDAZ(),Y $10
3:   STAZ(),Y $20
