# --- Labels (labels.asm) ---
# Label definitions & assignments, labels with colons, local labels, forward references

---
NAME: local_labels
INPUT:
 1: * = $0200
 2: global1:
 3: .local
 4:   NOP
 5:   BEQ .local
 6: global2:
 7: .local
 8:   NOP
 9:   BEQ .local
EXPECT_HEX: ea f0 fd ea f0 fd

---
NAME: forward_ref_absolute
INPUT:
 1: * = $0200
 2:   JMP target
 3:   NOP
 4: target:
 5:   RTS
EXPECT_HEX: 4c 04 02 ea 60

---
NAME: label_assignment
INPUT:
 1: * = $0200
 2: addr: = $1234
 3:   LDA addr
 4:   STA addr
EXPECT_HEX: ad 34 12 8d 34 12

---
NAME: label_not_found
INPUT:
 1: * = $0200
 2:   LDA undefined
EXPECT_ERROR: 1
EXPECT_LINE: 2
EXPECT_MSG: Label not found

---
NAME: no_global_for_local
INPUT:
 1: * = $0200
 2: .local
 3:   NOP
EXPECT_ERROR: 3
EXPECT_LINE: 2
EXPECT_MSG: No global label for local

---
NAME: duplicate_label

EXPECT_LINE: 4
